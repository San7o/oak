198 std::string formatted = std::vformat(fmt, std::make_format_args(args...));
199
200 std::string output = this->formatter(level, flags, file, line, formatted);
201
202 for (auto& writer : writers)
203 {
204 writer->write(output);
205 }
206 }

261 static inline const Formatter default_formatter =
262 [](enum Level level, int flags,
263 const char* file,
264 int line,
265 const std::string& log)
266 {
267 // It ain't pretty, but it does the job
268
269 std::string output;
270
271 bool json = false;
272 bool do_color = false;
273 if (flags & (unsigned int) Flags::Json)
274 json = true;
275 if (flags & (unsigned int) Flags::Color)
276 do_color = true;
277
278 if (json) output += "{ ";
279
280 if (flags & (unsigned int) Flags::Level)
281 {
282 if (json)
283 output += "\"level\": \"" + level_to_string(level) + "\", ";
284 else
285 output += "[ " + level_to_string(level) + " ] ";
286 }
287 if (flags & (unsigned int) Flags::Date)
288 {
289 auto now = std::chrono::system_clock::now();
290 auto now_time_t = std::chrono::system_clock::to_time_t(now);
291 std::tm now_tm = *std::localtime(&now_time_t);
292 std::ostringstream oss;
293 if (json)
294 oss << "\"date\": \"" << std::put_time(&now_tm, "%Y-%m-%d") << "\", ";
295 else
296 oss << "[ " << std::put_time(&now_tm, "%Y-%m-%d") << " ] ";
297 output += oss.str();
298 }
299 if (flags & (unsigned int) Flags::Time)
300 {
301 auto now = std::chrono::system_clock::now();
302 auto now_time_t = std::chrono::system_clock::to_time_t(now);
303 std::tm now_tm = *std::localtime(&now_time_t);
304 std::ostringstream oss;
305 if (json)
306 oss << "\"time\": \"" << std::put_time(&now_tm, "%H:%M:%S") << "\", ";
307 else
308 oss << "[ " << std::put_time(&now_tm, "%H:%M:%S") << " ] ";
309 output += oss.str();
310 }
311 if (flags & (unsigned int) Flags::Pid)
312 {
313 std::string pid = std::to_string(getpid());
314 if (json) output += "\"pid\": " + pid + ", ";
315 else output += "[ " + pid + " ] ";
316 }
317 if (flags & (unsigned int) Flags::Tid)
318 {
319 std::ostringstream oss;
320 oss << std::this_thread::get_id();
321 std::string tid = oss.str();
322 if (json) output += "\"tid\": " + tid + ", ";
323 else output += "[ " + tid + " ] ";
324 }
325 if (flags & (unsigned int) Flags::File)
326 {
327 if (json) output += "\"file\": \"" + std::string(file) + "\", ";
328 else output += "[ " + std::string(file) + " ] ";
329 }
330
331 if (flags & (unsigned int) Flags::Line)
332 {
333 if (json) output += "\"line\": " + std::to_string(line) + ", ";
334 else output += "[ " + std::to_string(line) + " ] ";
335 }
336
337 if (json) output += "\"log\": \"";
338
339 output += log;
340
341 if (json) output += "\" }";
342 output += '\n';
343
344 if (do_color)
345 return colorize(level, output);
346 return output;
347 };
