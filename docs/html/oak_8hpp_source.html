214template <typename... Args>
215std::string constexpr log_to_string(const level &lvl, const std::string &fmt,
216 Args &&...args)
217{
218 auto flags = get_flags();
219 bool json = false;
220 if (flags & static_cast<long unsigned int>(flags::json))
221 json = true;
222 std::string prefix = "";
223 if (flags > 0 && !json)
224 {
225 prefix += "[ ";
226 }
227 if (json)
228 {
229 prefix += "{ ";
230 }
231 if (flags & static_cast<long unsigned int>(flags::level))
232 {
233 if (json)
234 prefix += std::vformat("\"level\": \"{}\"", std::make_format_args(lvl));
235 else
236 prefix += std::vformat("level={} ", std::make_format_args(lvl));
237 }
238 if (flags & static_cast<long unsigned int>(flags::date))
239 {
240 auto now = std::chrono::system_clock::now();
241 auto now_time_t = std::chrono::system_clock::to_time_t(now);
242 std::tm now_tm = *std::localtime(&now_time_t);
243 std::ostringstream oss;
244 if (json)
245 oss << ", \"date\": \"" << std::put_time(&now_tm, "%Y-%m-%d") << "\"";
246 else
247 oss << "date=" << std::put_time(&now_tm, "%Y-%m-%d") << " ";
248 prefix += oss.str();
249 }
250 if (flags & static_cast<long unsigned int>(flags::time))
251 {
252 auto now = std::chrono::system_clock::now();
253 auto now_time_t = std::chrono::system_clock::to_time_t(now);
254 std::tm now_tm = *std::localtime(&now_time_t);
255 std::ostringstream oss;
256 if (json)
257 oss << ", \"time\": \"" << std::put_time(&now_tm, "%H:%M:%S") << "\"";
258 else
259 oss << "time=" << std::put_time(&now_tm, "%H:%M:%S") << " ";
260 prefix += oss.str();
261 }
262 if (flags & static_cast<long unsigned int>(flags::pid))
263 {
264 std::string pid = std::to_string(getpid());
265 if (json)
266 prefix += ", \"pid\": " + pid;
267 else
268 prefix += std::vformat("pid={} ", std::make_format_args(pid));
269 }
270 if (flags & static_cast<long unsigned int>(flags::tid))
271 {
272 std::ostringstream oss;
273 oss << std::this_thread::get_id();
274 std::string tid = oss.str();
275 if (json)
276 prefix += ", \"tid\": " + tid;
277 else
278 prefix += std::vformat("tid={} ", std::make_format_args(tid));
279 }
280
281 if (flags - static_cast<long unsigned int>(flags::json) > 0 && !json)
282 prefix += "] ";
283 if (flags - static_cast<long unsigned int>(flags::json) > 0 && json)
284 prefix += ", ";
285 try
286 {
287 std::string formatted_string =
288 std::vformat(fmt, std::make_format_args(std::forward<Args>(args)...));
289 if (json)
290 return std::vformat("{}\"message\": \"{}\" }}\n",
291 std::make_format_args(prefix, formatted_string));
292 return std::vformat("{}{}\n",
293 std::make_format_args(prefix, formatted_string));
294 }
295 catch (const std::exception &e)
296 {
297 return "";
298 }
299}

341template <typename... Args>
342void log(const level &lvl, const std::string &fmt, Args &&...args)
343{
344 if (get_level() > lvl)
345 return;
346 std::string formatted_string = log_to_string(lvl, fmt, args...);
347 if (get_flags() & static_cast<long unsigned int>(flags::color))
348 log_to_stdout(apply_color(lvl, formatted_string));
349 else
350 log_to_stdout(formatted_string);
351 if (is_file_open())
352 log_to_file(formatted_string);
353#ifdef OAK_USE_SOCKETS
354 if (logger::log_socket > 0)
355 log_to_socket(formatted_string);
356#endif
357}
